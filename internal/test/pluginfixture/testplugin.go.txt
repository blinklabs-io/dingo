// Test plugin built by the plugin package tests from a module whose only
// dependency is the sdk. It serves the standard gRPC health service as the
// capability contract.
package main

import (
	"bufio"
	"context"
	"flag"
	"fmt"
	"os"
	"os/signal"
	"syscall"

	sdk "github.com/blinklabs-io/dingo/sdk/plugin"
	healthpb "google.golang.org/grpc/health/grpc_health_v1"
)

type health struct {
	healthpb.UnimplementedHealthServer
}

func (health) Check(
	_ context.Context,
	req *healthpb.HealthCheckRequest,
) (*healthpb.HealthCheckResponse, error) {
	if req.GetService() == "crash" {
		go os.Exit(7)
	}
	return &healthpb.HealthCheckResponse{
		Status: healthpb.HealthCheckResponse_SERVING,
	}, nil
}

// Watch blocks until the caller goes away, which keeps a graceful stop from
// finishing.
func (health) Watch(
	_ *healthpb.HealthCheckRequest,
	stream healthpb.Health_WatchServer,
) error {
	if err := stream.Send(&healthpb.HealthCheckResponse{
		Status: healthpb.HealthCheckResponse_SERVING,
	}); err != nil {
		return err
	}
	<-stream.Context().Done()
	return stream.Context().Err()
}

func main() {
	capability := flag.String("capability", "example.widget/v0", "")
	exitAtStart := flag.Int("exit-at-start", 0, "")
	ignoreTerm := flag.Bool("ignore-term", false, "")
	flag.Parse()
	if *exitAtStart != 0 {
		// Take the handshake first so the exit, not a refused write, is what
		// the node observes.
		_, _ = bufio.NewReader(os.Stdin).ReadString('\n')
		fmt.Fprintln(os.Stderr, "testplugin: refusing to start")
		os.Exit(*exitAtStart)
	}
	if *ignoreTerm {
		signal.Ignore(syscall.SIGTERM)
	}
	err := sdk.Serve(
		sdk.Manifest{Name: "testplugin", Version: "1.2.3"},
		sdk.Implementation{
			Capability: *capability,
			Service:    &healthpb.Health_ServiceDesc,
			Impl:       health{},
		},
	)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
